# Проверка конфигурации

Единственное место хранения автоматической диагностики конфигурации интеграции с ЕГИСЗ. Запрос выполняется к базе данных МИС и возвращает построчный результат `CHECK_NAME` / `CHECK_RESULT`. Значение `ОК` означает, что параметр заполнен и прошёл проверку; любое другое значение указывает на конкретную незаполненную или некорректную настройку.

Ссылки на этот запрос — из [чек-листа готовности](../testing/readiness.mdx) и [алгоритма диагностики](./index.mdx). Не копировать сам запрос на другие страницы.

## Проверяемые блоки

### Данные организации

- Адрес callback-сервиса (`FIR_EMD_CALLBACK`).
- ОГРН юридического лица (13 символов; 15 — для ИП).
- Юридический и фактический адрес, телефон организации.
- OID юридического лица, код территории.
- Сведения о лицензии юридического лица.
- Временная зона филиала (значение по умолчанию `0` считается незаполненным).

### Данные подразделений

- OID подразделения в ФРМО — для каждого подразделения (`cashref`).
- Наименование подразделения в ФРМО, используемое при формировании РЭМД.
- Медицинские специализации действующих отделений.

### Данные сотрудников

- Наличие действующего сертификата УКЭП (тип 1, статус 3, срок действия не истёк) у сотрудников с заполненным СНИЛС.
- Наличие специальности в отделении у сотрудников с действующим сертификатом УКЭП.

### Инфраструктура

- Наличие триггеров РЭМД на таблице `TREATPLACESIGNDET` (`FIR_EMD_TREATPLACESIGN%`).
- Наличие записи о базе данных интеграции (`filial_base`, `grpid = 9924`, `basetype = 1`).
- Наличие записи о медиа-базе данных интеграции (`filial_base`, `grpid = 9924`, `basetype = 2`).

### НСИ

- Сопоставление хотя бы одной услуги (`wschema`) с номенклатурой Минздрава (`officialcodeid`).

## Проверочный запрос

<details>
<summary>Полный SQL-запрос (18 проверок)</summary>

```sql title="egisz-check.sql"
-- 1. Проверка адреса callback сервиса
SELECT
    'Адрес callback сервиса' AS CHECK_NAME,
    iif(coalesce(keyvalue,'') = '', 'В настройках не заполнен', 'ОК') AS CHECK_RESULT
FROM m_config WHERE keyname = 'FIR_EMD_CALLBACK'

UNION ALL

-- 2. Проверка ОГРН юридического лица
SELECT
    'ОГРН юридического лица' AS CHECK_NAME,
    iif(coalesce(j.ogrn, '') = '', 'Не заполнен', iif(char_length(j.ogrn) <> 13, 'Заполнен некорректно (кроме ИП - 15 символов)', 'ОК'))
FROM jpersons j
JOIN filials f ON f.jid = j.jid
WHERE f.filid = (SELECT keyvalue FROM m_config WHERE upper(keyname) = upper('filial'))

UNION ALL

-- 3. Проверка юридического адреса
SELECT
    'Юридический адрес' AS CHECK_NAME,
    iif(coalesce(jpau.klcode, '') = '', 'Не указан', 'ОК')
FROM jpersons j
JOIN filials f ON f.jid = j.jid
LEFT JOIN jpaddress jpau ON j.jid = jpau.recid AND jpau.addrtype = 1
WHERE f.filid = (SELECT keyvalue FROM m_config WHERE upper(keyname) = upper('filial'))

UNION ALL

-- 4. Проверка фактического адреса
SELECT
    'Фактический адрес' AS CHECK_NAME,
    iif(coalesce(jpaf.klcode, '') = '', 'Не указан', 'ОК')
FROM jpersons j
JOIN filials f ON f.jid = j.jid
LEFT JOIN jpaddress jpaf ON j.jid = jpaf.recid AND jpaf.addrtype = 0
WHERE f.filid = (SELECT keyvalue FROM m_config WHERE upper(keyname) = upper('filial'))

UNION ALL

-- 5. Проверка телефона организации
SELECT
    'Телефон организации' AS CHECK_NAME,
    iif(coalesce(jpau.phonetext, '') = '', 'Не указан', 'ОК')
FROM jpersons j
JOIN filials f ON f.jid = j.jid
LEFT JOIN jpaddress jpau ON j.jid = jpau.recid AND jpau.addrtype = 1
WHERE f.filid = (SELECT keyvalue FROM m_config WHERE upper(keyname) = upper('filial'))

UNION ALL

-- 6. Проверка OID юридического лица
SELECT
    'OID юридического лица' AS CHECK_NAME,
    iif(coalesce(j.fir_oid,'') = '', 'Не указан', 'ОК')
FROM jpersons j
JOIN filials f ON f.jid = j.jid
WHERE f.filid = (SELECT keyvalue FROM m_config WHERE upper(keyname) = upper('filial'))

UNION ALL

-- 7. Проверка кода территории юридического лица
SELECT
    'Код территории юр. лица' AS CHECK_NAME,
    iif(coalesce(j.kodter,'') = '', 'Не указана', 'ОК')
FROM jpersons j
JOIN filials f ON f.jid = j.jid
WHERE f.filid = (SELECT keyvalue FROM m_config WHERE upper(keyname) = upper('filial'))

UNION ALL

-- 8. Проверка лицензии юридического лица
SELECT
    'Лицензия юридического лица' AS CHECK_NAME,
    iif(coalesce(j.licnumber,'') = '', 'Не указаны сведения', 'ОК')
FROM jpersons j
JOIN filials f ON f.jid = j.jid
WHERE f.filid = (SELECT keyvalue FROM m_config WHERE upper(keyname) = upper('filial'))

UNION ALL

-- 9. Проверка временной зоны филиала
SELECT
    'Временная зона филиала' AS CHECK_NAME,
    iif(f.timezone = 0, 'Установлено значение по умолчанию (0)', 'ОК')
FROM filials f
WHERE f.filid = (SELECT keyvalue FROM m_config WHERE upper(keyname) = upper('filial'))

UNION ALL

-- 10. Проверка OID подразделений
SELECT
    'OID подразделения в ФРМО (' || trim(csh.cashname) || ')' AS CHECK_NAME,
    iif(coalesce(csh.fir_oid,'') = '', 'Не указан', 'ОК')
FROM cashref csh
WHERE csh.filial = (SELECT keyvalue FROM m_config WHERE upper(keyname) = upper('filial'))

UNION ALL

-- 11. Проверка наименования подразделений для РЭМД
SELECT
    'Наименование подразделения в ФРМО (' || trim(csh.cashname) || ')' AS CHECK_NAME,
    iif(coalesce(csh.fir_name,'') = '', 'Не указано', 'ОК')
FROM cashref csh
WHERE csh.filial = (SELECT keyvalue FROM m_config WHERE upper(keyname) = upper('filial'))

UNION ALL

-- 12. Проверка наличия триггеров
SELECT
    'Наличие триггеров РЭМД' AS CHECK_NAME,
    iif(exists(select RDB$TRIGGER_NAME from RDB$TRIGGERS where RDB$TRIGGER_NAME containing 'FIR_EMD_TREATPLACESIGN' and upper(RDB$RELATION_NAME) = upper('TREATPLACESIGNDET')), 'OK', 'Отсутствуют')
FROM RDB$DATABASE

UNION ALL

-- 13. Проверка записи о БД для интеграции с РЭМД
SELECT
    'Запись о БД интеграции' AS CHECK_NAME,
    iif(exists(select * from filial_base where grpid = 9924 and basetype = 1), 'OK', 'Отсутствует')
FROM RDB$DATABASE

UNION ALL

-- 14. Проверка записи о медиа БД для интеграции с РЭМД
SELECT
    'Запись о медиа БД интеграции' AS CHECK_NAME,
    iif(exists(select * from filial_base where grpid = 9924 and basetype = 2), 'OK', 'Отсутствует')
FROM RDB$DATABASE

UNION ALL

-- 15. Сопоставление услуг с номенклатурой Минздрава
SELECT
    'Код услуги по НСИ Минздрава' AS CHECK_NAME,
    iif(exists(select * from wschema w where coalesce(w.officialcodeid, 0) > 0), 'ОК', 'Не заполнен ни для одной услуги')
FROM RDB$DATABASE

UNION ALL

-- 16. Список медицинских специализаций отделения
SELECT
    'Специализации отделения (' || trim(d.depname) || ')' AS CHECK_NAME,
    iif(coalesce(d.idmsplist, '') = '', 'Не указаны', 'ОК')
FROM departments d
WHERE d.filial = (SELECT keyvalue FROM m_config WHERE upper(keyname) = upper('filial'))
  AND (d.disdate is null or d.disdate > current_date)

UNION ALL

-- 17. Действующий сертификат УКЭП сотрудника
SELECT
    'Сертификат УКЭП (' || trim(doc.fullname) || ')' AS CHECK_NAME,
    iif(exists(select * from certdoctor cd
               join certificate c on c.certid = cd.certid
               where cd.dcode = doc.dcode
                 and c.certtype = 1
                 and c.certstate = 3
                 and (c.validto is null or c.validto >= current_date)), 'ОК', 'Действующий сертификат отсутствует')
FROM doctor doc
WHERE doc.filial = (SELECT keyvalue FROM m_config WHERE upper(keyname) = upper('filial'))
  AND coalesce(doc.locked, 0) <> 1
  AND coalesce(doc.snils, '') <> ''

UNION ALL

-- 18. Специальность в отделении у подписывающих сотрудников
SELECT
    'Специальность в отделении (' || trim(doc.fullname) || ')' AS CHECK_NAME,
    iif(exists(select * from doctdepartlinks ddl
               where ddl.dcode = doc.dcode
                 and coalesce(ddl.idmsp, 0) > 0
                 and (ddl.bdate is null or ddl.bdate <= current_date)
                 and (ddl.fdate is null or ddl.fdate >= current_date)), 'ОК', 'Не указана')
FROM doctor doc
WHERE doc.filial = (SELECT keyvalue FROM m_config WHERE upper(keyname) = upper('filial'))
  AND coalesce(doc.locked, 0) <> 1
  AND exists(select * from certdoctor cd
             join certificate c on c.certid = cd.certid
             where cd.dcode = doc.dcode
               and c.certtype = 1
               and c.certstate = 3
               and (c.validto is null or c.validto >= current_date))
```
